def aggregate(rows,policy,now):
 candidates={};info={}
 for tid,cfg in CAT.items():
  r=[x for x in rows if x['theme']==tid and x['policy_keep'][policy]]
  if not r:continue
  weighted=lambda x:(x['policy_D_raw'] if policy=='D' else x['raw_score'])*x['cluster_weight']
  total=sum(weighted(x) for x in r);n=len(r);sources={x['source'] for x in r};cats={x['category'] for x in r}
  recent=sum(bool(x.get('published_dt')) and (now-datetime.fromisoformat(x['published_dt'])).total_seconds()<21600 for x in r)
  conf=min(95,int(total*2+n*4+min(len(sources)*2,10)+(8 if len(cats)>=2 else 0)+min(recent*3,9)))
  gen=sum(len(x['generic']) for x in r);kh=sum(len(x['keyword_hits']) for x in r);ratio=gen/max(kh,1)
  if ratio>.6 and conf>=25:conf=max(1,int(conf*(1-min(.30,.15+(ratio-.6)*.375))))
  breadth=min(len(cfg['related_industries']),max(1,n))
  gates=n>=2 and len(sources)>=2 and breadth>=2 and conf>=20 and total>=1
  top=sorted(r,key=weighted,reverse=True)[:5]
  info[tid]=dict(eligible_pairs=n,total_weight=total,confidence_pre_competition=conf,evidence_count=len(sources),top=[x['cluster_or_observation_id'] for x in top],passed_precompetition=gates)
  if gates:candidates[tid]=conf
 out={}
 for tid,conf in candidates.items():
  penalty=min(.4,sum(.15 for ct in CAT[tid].get('competing_themes',[]) if candidates.get(ct,-1)>conf));final=max(1,int(conf*(1-penalty))) if penalty else conf
  if final>=20:out[tid]=dict(info[tid],confidence=final,competition_penalty=round(penalty,2))
 return out,info
